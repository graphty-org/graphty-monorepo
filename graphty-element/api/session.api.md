// @public
export type GraphtyErrorCode =
/**
* A command or a call is malformed: a missing required field, or two fields that may not be
* set together. `details` name the fields.
* For saved views, `details` are `{ reason: "empty-name" }` for an empty name,
* `{ reason: "name-taken", name }` for a rename onto a name in use, and
* `{ missing, available }` for a rename, move or removal of a name that is not saved.
* The caller fixes the call; retrying the same call always fails the same way.
*/
"E_BAD_COMMAND"
/**
* A query or pattern string does not parse. `details` carry the character offset and the
* token that failed. The caller corrects the expression, usually in front of a person who
* typed it.
*/
| "E_BAD_QUERY"
/**
* A style layer's shape is wrong: an unknown key, a value of the wrong type, or a nested
* member where none belongs. `details.path` names the offending path inside the layer, so
* an editor can point at it. A wrong shape is never silently accepted.
*/
| "E_BAD_LAYER"
/**
* A style layer's selector does not parse. `details.reason` is a stable code naming the
* mistake (such as `number-needs-backticks`), so a consumer writes its own words without
* reading the message; an expression's refusal also carries the character offset
* (`details.position`). A `number-needs-backticks` refusal also carries `details.suggestion`,
* the selector with every bare number put between backticks (`` minutes >= `10` ``). The
* caller corrects the selector; the layer is not added.
*/
| "E_BAD_SELECTOR"
/**
* A computed-attribute formula does not parse. `details` carry the character offset and the
* token that failed. The caller corrects the formula.
*/
| "E_BAD_FORMULA"
/**
* A selector is empty, which would match every node and every edge. An algorithm layer that
* paints the whole graph erases every layer beneath it, so an empty selector is refused
* rather than interpreted. The caller writes a selector that names what it means to paint.
*/
| "E_SELECTOR_EMPTY"
/**
* A universal selector is combined with an encoding bound to a run that paints a value for a
* missing result (`missing: { value }`), which would ask the element to paint elements the run
* never measured. Without `missing`, such a layer is accepted: an element with no result is
* not painted. The caller scopes the layer to the
* elements that carry the run's result: the problem's `scope` is that selector.
*/
| "E_UNSCOPED_RUN_ENCODING"
/**
* An encoding names a scale that is not registered. `details.available` lists the registered
* scale names. The caller picks one of those or registers its own scale first.
*/
| "E_UNKNOWN_SCALE"
/**
* A style or encoding names a channel that does not exist. `details.available` lists the
* channels for that target. Usually a spelling: a misspelled channel must never be a silent
* no-op.
*/
| "E_UNKNOWN_CHANNEL"
/**
* An option name is not one this algorithm, layout, format or exporter accepts.
* `details.available` lists the valid names and `details.candidates` the nearest few, so the
* caller can show "did you mean".
*/
| "E_UNKNOWN_OPTION"
/**
* An option is known but its value is outside the permitted range or not one of the
* permitted values, or a required option is missing. `details` carry `option`, the value that
* was passed (absent when none was) and, where known, the range or the value list. An
* algorithm's or layout's own `E_BAD_OPTION` arrives as this code. The caller clamps or
* corrects the value.
*/
| "E_OPTION_RANGE"
/**
* An expression, selector or formula names an attribute that nothing in the session answers.
* `details.candidates` carry the nearest three names. Also a layout option that names a node
* column the graph does not hold (`multipartite`'s `subsets`); `details` then carry `option`
* and `column`. The caller corrects the name, or loads the data that would define it.
*/
| "E_UNKNOWN_ATTRIBUTE"
/**
* A command, API member, saved document or attribute asked for something whose experiment is
* off: `details` (a project open's problem: `params`) carry `feature`, the experiment's id, and
* `key`, what was asked for. The caller switches the experiment on (`features="..."`) or leaves
* that part out.
*/
| "E_FEATURE_OFF"
/**
* A timeline call needs a time field and none is designated: `session.time.window`, `play`,
* `stepBy` or `seek` with no attribute given. The caller designates one with
* `session.time.setField` (or the `node-time-path` / `edge-time-path` attributes).
* Beta: may change in a minor release.
*/
| "E_NO_TIME_FIELD"
/**
* An algorithm key is not registered. `details.available` lists the registered keys. The
* caller picks a key from the catalog or registers a plugin that provides it.
*/
| "E_UNKNOWN_ALGORITHM"
/**
* A layout name is not registered. `details.available` lists the registered layout names.
* An unknown layout fails loudly rather than leaving the graph in whatever arrangement it
* already had.
*/
| "E_UNKNOWN_LAYOUT"
/**
* A data format id is not one the element can import or export. `details.available` lists
* the formats and which direction each supports. The caller picks a supported format or
* registers a plugin for its own.
*/
| "E_UNKNOWN_FORMAT"
/**
* A binding, an `encode` call or a saved document names a palette nothing registered.
* `details.available` lists the known palette ids and `details.candidates` the nearest few.
* The caller picks a palette from the catalog or registers its own first.
*/
| "E_UNKNOWN_PALETTE"
/**
* A layer names a shader the catalog does not list, in a use on `node.shaders` or
* `edge.shaders` or a parameter's address. `details.candidates` lists the ids that list takes.
* Since 3.36.0.
*/
| "E_UNKNOWN_SHADER"
/**
* A layer names a shader where it cannot draw: an edge shader on `node.shaders`
* (`shader.target-mismatch`), or a use at a place the shader does not allow
* (`shader.place-unsupported`); the refused problem's `fact` carries the code.
* Since 3.36.0.
*/
| "E_SHADER_KIND"
/**
* A layer sets or binds a parameter the shader does not declare, or binds one per element
* that it does not declare `varies`. `details.candidates` lists the declared parameters.
* Since 3.36.0.
*/
| "E_UNKNOWN_SHADER_PARAM"

/**
* The members of a scope, usually a kept set: `{ match: "member", of: { set: id } }`. The
* layer follows the set: a redefinition repaints exactly the elements that moved. A removed
* set paints from its kept record, so removing a set never blanks a layer; a scope that
* cannot be evaluated paints nothing and never throws.
*/
| {
    match: "member";
    of: Scope;
}
/**
* The elements a rule tree holds, re-evaluated as the data changes: what a rule set holds,
* without keeping a set. `edges` is how the edges are read beside a rule about nodes (default
* `clipped`, the edges between its nodes, as the visibility filter reads it). "Top N by" any
* number is a `threshold` leaf: `{ kind: "threshold", path: "data.age", top: 10 }`.
*/
| {
    match: "rule";
    where: RuleTree;
    edges?: EdgeReading;
}
/**
* The elements a reader is pointing at now: selected (`"selected"`, the selection's nodes and
* edges), under the pointer (`"hovered"`, one node at most), drawn while a held filter
* condition would remove it (`"ghosted"`), outside the time window (`"outsideWindow"`:
* a layer selecting it decides how it looks: hidden, faint, or as any other), or pinned in place
* (`"pinned"`, nodes only, as `session.positions.pinned` holds them). View state: the layer
* follows each change with no step. The same elements as a rule: a {@link StateLeaf}, which
* `selectorRule` reads this selector as with its `states` option.
* @since 3.26.0
*/
| {
    match: "state";
    state: ElementState;
}
/**
* What the reader is pointing at: the node under the pointer (`reach: "node"`, the default),
* or that node, its neighbors and the edges that touch it (`reach: "neighbors"`). With
* `invert: true`, everything else -- but only while a node is hovered, so the layer paints
* nothing at rest. Edges: the edges that touch the hovered node, for either reach. View
* state: the layer follows the pointer with no undo step. `session.selection.hovered` says
* which node it is. With `orSelected: true`, while no node is hovered a lone selected node
* stands in for the hovered one, so a look that follows the pointer stays on the node the
* reader clicked; several selected nodes, or none, and the layer paints as at rest.
*
* ```ts
* // Fade what the pointed-at node is not linked to.
* session.styles.add({ name: "Dim the rest", target: "node",
*     selector: { match: "hover", reach: "neighbors", invert: true }, set: { "node.opacity": 0.25 } });
* ```
*/
| {
    match: "hover";
    reach?: "node" | "neighbors";
    invert?: boolean;
    orSelected?: boolean;
}
/**
* The elements of some types, under the `source-steps` experiment: those whose one type
* (the readable path `graphty.type`) is listed, such as the accounts of a graph of accounts and
* merchants. Refused with `E_FEATURE_OFF` while the experiment is off.
*
* ```ts
* session.styles.add({ name: "Accounts", target: "node",
*     selector: { match: "type", types: ["account"] }, set: { "node.color": "#4c6ef5" } });
* ```
* Experimental: may change or be removed in a minor release.
*/
| {
    match: "type";
    types: readonly string[];
};
