// Gemstone: flat facets, rainbow fire from per-channel refraction, twinkling glints.

float studioBox(vec3 d, vec3 s, vec2 size, float soft) {
    vec3 t = normalize(cross(s, vec3(0.0, 1.0, 0.0)));
    vec3 b = cross(t, s);
    float k = dot(d, s);
    if (k <= 0.0) return 0.0;
    vec2 p = vec2(dot(d, t), dot(d, b)) / k;
    vec2 q = abs(p) - size;
    return 1.0 - smoothstep(-soft, soft, max(q.x, q.y));
}
// horizon: the half-width of the blend from floor to sky; studio() is the crisp one.
vec3 studioSoft(vec3 d, float horizon) {
    d = normalize(d);
    vec3 sky = mix(vec3(0.62, 0.68, 0.78), vec3(0.10, 0.13, 0.24), smoothstep(0.0, 0.85, d.y));
    vec3 ground = mix(vec3(0.22, 0.20, 0.19), vec3(0.03, 0.03, 0.035), smoothstep(0.0, -0.5, d.y));
    vec3 c = mix(ground, sky, smoothstep(-horizon, horizon, d.y));
    c += vec3(5.0, 4.8, 4.5) * studioBox(d, normalize(vec3(0.55, 0.65, -0.5)), vec2(0.35, 0.22), 0.04);
    c += vec3(2.2, 2.6, 3.4) * studioBox(d, normalize(vec3(-0.7, 0.35, 0.6)), vec2(0.18, 0.5), 0.05);
    c += vec3(1.6) * studioBox(d, normalize(vec3(0.0, 0.15, 1.0)), vec2(0.9, 0.04), 0.03);
    return c;
}
vec3 studio(vec3 d) { return studioSoft(d, 0.03); }
// A filmic tone curve for light that runs past 1.
vec3 studioTone(vec3 c) { return 1.0 - exp(-c * 1.15); }

// The i-th of count directions spread evenly over the sphere (a Fibonacci lattice).
vec3 gemFacet(float i, float count) {
    float y = 1.0 - 2.0 * (i + 0.5) / count;
    float r = sqrt(max(0.0, 1.0 - y * y));
    float a = i * 2.39996323;
    return vec3(cos(a) * r, y, sin(a) * r);
}
// The facet nearest to d (object space): xyz its normal, w the margin to the next facet.
vec4 facetOf(vec3 d, float count) {
    float best = -2.0;
    float second = -2.0;
    vec3 bn = d;
    for (int i = 0; i < 64; i++) {
        if (float(i) >= count) break;
        vec3 f = gemFacet(float(i), count);
        float k = dot(d, f);
        if (k > best) { second = best; best = k; bn = f; } else if (k > second) { second = k; }
    }
    return vec4(bn, best - second);
}
vec3 toObj(vec3 v, float a, float b) { v.xz = rot2(a) * v.xz; v.xy = rot2(b) * v.xy; return v; }
vec3 fromObj(vec3 v, float a, float b) { v.xy = rot2(-b) * v.xy; v.xz = rot2(-a) * v.xz; return v; }
// Dark surroundings and hot lights make the fire read.
vec3 env(vec3 d) { vec3 e = studio(d); return pow(e, vec3(1.5)) * 1.4 + vec3(0.05, 0.045, 0.07); }

// One color channel's path through the stone (unit sphere, object rotation ra, rb): refract in,
// bounce off two back facets, refract out.
vec3 fire(vec3 p, vec3 iw, vec3 facetN, float eta, float ra, float rb, float count) {
    vec3 t = refract(iw, facetN, 1.0 / eta);
    for (int k = 0; k < 2; k++) {
        p = normalize(p + t * max(-2.0 * dot(p, t), 0.0));
        vec3 back = fromObj(facetOf(toObj(p, ra, rb), count).xyz, ra, rb);
        vec3 out1 = refract(t, -back, eta);
        if (k == 1 && dot(out1, out1) > 1e-4) return env(out1) * 0.9;
        t = reflect(t, back);
    }
    return env(t) * 0.7;
}

vec4 shade(NodeInput n) {
    vec3 nv = n.normal;
    vec3 iw = n.viewDir;
    float count = float(facets);
    float ra = n.seed * 40.0 + n.time * turn;
    float rb = n.seed * 17.0;
    vec4 fo = facetOf(toObj(nv, ra, rb), count);
    vec3 facetN = fromObj(fo.xyz, ra, rb);
    vec3 iorC = ior + dispersion * vec3(-1.0, 0.0, 1.0);
    // One ray per color channel is the expensive part: medium traces red and blue and blends
    // green, low traces one ray for all three.
    int level = Q(0, 1, 2);
    vec3 inside;
    if (level == 0) {
        inside = fire(nv, iw, facetN, iorC.y, ra, rb, count);
    } else if (level == 1) {
        vec3 fr = fire(nv, iw, facetN, iorC.x, ra, rb, count);
        vec3 fb = fire(nv, iw, facetN, iorC.z, ra, rb, count);
        inside = vec3(fr.r, 0.5 * (fr.g + fb.g), fb.b);
    } else {
        inside = vec3(
            fire(nv, iw, facetN, iorC.x, ra, rb, count).r,
            fire(nv, iw, facetN, iorC.y, ra, rb, count).g,
            fire(nv, iw, facetN, iorC.z, ra, rb, count).b
        );
    }

    vec3 body = pow(n.color.rgb, vec3(2.2)) * 2.2;
    float ndv = max(dot(facetN, -iw), 0.0);
    float f0 = pow((ior - 1.0) / (ior + 1.0), 2.0);
    float fres = f0 + (1.0 - f0) * pow(1.0 - ndv, 5.0);
    vec3 c = inside * body * (1.0 - fres) + env(reflect(iw, facetN)) * fres;
    float glint = pow(max(dot(reflect(iw, facetN), lightDir), 0.0), 200.0);
    float twinkle = 0.5 + 0.5 * sin(n.time * 7.0 + hash13(fo.xyz * 91.0 + n.seed) * 60.0);
    c += sparkle * glint * twinkle * vec3(6.0);
    // Facet edges catch a little light.
    c += vec3(0.05) * (1.0 - smoothstep(0.0, 0.012, fo.w));
    return vec4(studioTone(c), 1.0);
}

vec4 shadeFar(NodeInput n) {
    vec3 nv = n.normal;
    vec3 iw = n.viewDir;
    float ra = n.seed * 40.0 + n.time * turn;
    float rb = n.seed * 17.0;
    vec3 facetN = fromObj(facetOf(toObj(nv, ra, rb), 8.0).xyz, ra, rb);
    vec3 body = pow(n.color.rgb, vec3(2.2)) * 2.2;
    float ndv = max(dot(facetN, -iw), 0.0);
    float f0 = pow((ior - 1.0) / (ior + 1.0), 2.0);
    float fres = f0 + (1.0 - f0) * pow(1.0 - ndv, 5.0);
    vec3 c = env(refract(iw, facetN, 1.0 / ior)) * 0.6 * body * (1.0 - fres) + env(reflect(iw, facetN)) * fres;
    c += sparkle * pow(max(dot(reflect(iw, facetN), lightDir), 0.0), 60.0) * vec3(2.0);
    return vec4(studioTone(c), 1.0);
}
