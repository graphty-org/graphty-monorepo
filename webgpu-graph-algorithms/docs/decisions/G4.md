### tmp/p4/t14/append-session.mjs

```js
/**
 * Appends the LAST session of benchmarks/out/<class>.json to benchmarks/results/<class>.json (contract 6.4: a JSON array
 * of sessions ordered by date; the last one is the baseline). Refuses a software session (spec 11.7) and a session that
 * lacks one of the seven groups (bench-compare reads only the last session, so it must carry every row). docs/decisions/
 * G3.md appendix A with REQUIRED_GROUPS extended by layout-fr (P5) and layout-grid (P4-T14).
 *
 *   node tmp/p4/t14/append-session.mjs <out file> <results file>
 */
import { existsSync, readFileSync, writeFileSync } from "node:fs";

const REQUIRED_GROUPS = ["upload", "roundtrip", "layout-exact", "pagerank", "wcc", "layout-fr", "layout-grid"];
const [outFile, resultsFile] = process.argv.slice(2);
if (outFile === undefined || resultsFile === undefined) {
    console.error("usage: node tmp/p4/t14/append-session.mjs <out file> <results file>");
    process.exit(1);
}
const out = JSON.parse(readFileSync(outFile, "utf8"));
if (!Array.isArray(out) || out.length === 0) {
    console.error(`refusing: ${outFile} holds no session`);
    process.exit(1);
}
const last = out[out.length - 1];
if (last.gpu.software) {
    console.error(`refusing: the session of ${last.date} ran on a software adapter (${last.gpu.description})`);
    process.exit(1);
}
const groups = new Set(last.results.map((r) => r.group));
for (const g of REQUIRED_GROUPS) {
    if (!groups.has(g)) {
        console.error(
            `refusing: the session of ${last.date} lacks the ${g} group; run pnpm run bench with every group`,
        );
        process.exit(1);
    }
}
const sessions = existsSync(resultsFile) ? JSON.parse(readFileSync(resultsFile, "utf8")) : [];
if (sessions.some((s) => s.date === last.date)) {
    console.error(`refusing: a session dated ${last.date} is already in ${resultsFile}`);
    process.exit(1);
}
sessions.push(last);
sessions.sort((a, b) => a.date.localeCompare(b.date));
writeFileSync(resultsFile, `${JSON.stringify(sessions, null, 4)}\n`);
console.log(
    `appended the session of ${last.date} (${last.runnerClass}, software=${String(last.gpu.software)}, ` +
        `${last.results.length} results, groups ${[...groups].join(",")}) to ${resultsFile}; ${sessions.length} sessions now`,
);
```

/**
 * The Q-32 / R-24 knob sweep of the G4 record's section 7 (P4-T17): gridMax2D in {512, 1024, 2048} x extentFactor in
 * {4, 6, 8}; per setting the grid tier's GPU ms per iteration on a 1M-node / 10M-edge random graph (the layout-grid
 * protocol: an untimed clock warm-up burst, then the mean of KNOB_ITERATIONS timed step(1) calls' stats.msPerIteration),
 * and on clumpy1000 and isolated at 100k (fixture scale 5 and 100) the one-iteration exact-vs-grid error of spec 11.4
 * (the floored per-node relative error of the force after G7 against the exact tier's after K3, from the same start:
 * test/helpers/grid-parity.ts exactVsGrid and matchers.ts flooredRelError re-stated here because both reach the vitest
 * globals: the helper through sabotage.ts, the matcher through expect).
 *
 *   LD_LIBRARY_PATH=<the libEGL tree> GRAPHTY_GPU_REQUIRE=hardware pnpm exec tsx tmp/p4/knobs.ts [--small]
 *
 * --small runs 20k / 4k / 800 nodes instead (a lavapipe smoke of the script, never a number for the record).
 */
import { createForceAtlas2 } from "../../src/layouts/forceatlas2.js";
import { ForceSimulation } from "../../src/layouts/force-simulation.js";
import { createNodeGpuContext } from "../../src/node/index.js";
import type { ForceAtlas2Stats, GpuLayoutTuning } from "../../src/types/layout.js";
import type { ForceAtlas2Options } from "../../src/types/options.js";
import { randomEdges, snapshotOf } from "../../benchmarks/datasets.js";
import { warmClock } from "../../benchmarks/layout-exact.bench.js";
import { fixture } from "../../test/helpers/graphs.js";
import { seededScenePositions } from "../../test/oracle/forceatlas2.js";

const small = process.argv.includes("--small");
const GRID_MAX_2D = [512, 1024, 2048];
const EXTENT_FACTOR = [4, 6, 8];
const TIMED_NODES = small ? 20_000 : 1_000_000;
const KNOB_ITERATIONS = 20;
const WARM_MS = 500;
/** fixture() scale of the 20k-node family: 5 -> 100k nodes (the plan's "at 100k"). */
const PARITY_SCALE = small ? 0.2 : 5;
/** fixture() scale of isolated (giant 1,000 + 1% strays + 100 triangles): 100 -> 100,000 giant nodes. */
const ISOLATED_SCALE = small ? 0.8 : 100;
const FLOOR_FRACTION = 1e-3;
/** test/helpers/fa2-parity.ts BASE_OPTIONS (the parity options: scale 1, zero centre, paper mode). */
const BASE: ForceAtlas2Options = {
    dim: 2,
    scale: 1,
    center: [0, 0, 0],
    seed: 7,
    gravity: 1,
    scalingRatio: 2,
    jitterTolerance: 1,
    settleThreshold: 0,
    settleWindow: 10,
    maxIter: 1000,
    iterationsPerStep: 1,
    maxInFlight: 2,
};

type Sim = ForceSimulation<ForceAtlas2Options, ForceAtlas2Stats>;

/** test/helpers/matchers.ts flooredRelError without its expect: the spec 11.4 floored per-node relative error. */
function flooredRelError(
    gpu: ArrayLike<number>,
    cpu: ArrayLike<number>,
    floorFraction: number,
): { rms: number; p99: number; max: number } {
    if (gpu.length !== cpu.length) {
        throw new Error("flooredRelError: length");
    }
    const n = Math.floor(cpu.length / 3);
    const norms = new Float64Array(n);
    let largest = 0;
    for (let i = 0; i < n; i++) {
        norms[i] = Math.hypot(cpu[3 * i], cpu[3 * i + 1], cpu[3 * i + 2]);
        largest = Math.max(largest, norms[i]);
    }
    const floor = floorFraction * largest;
    const errors = new Float64Array(n);
    let sumSq = 0;
    let max = 0;
    for (let i = 0; i < n; i++) {
        const numerator = Math.hypot(
            gpu[3 * i] - cpu[3 * i],
            gpu[3 * i + 1] - cpu[3 * i + 1],
            gpu[3 * i + 2] - cpu[3 * i + 2],
        );
        const denominator = Math.max(norms[i], floor);
        const error = denominator > 0 ? numerator / denominator : numerator > 0 ? Number.POSITIVE_INFINITY : 0;
        errors[i] = error;
        sumSq += error * error;
        max = Math.max(max, error);
    }
    const sorted = Float64Array.from(errors).sort();
    const p99 = n === 0 ? 0 : sorted[Math.min(n - 1, Math.ceil(0.99 * n) - 1)];
    return { rms: n === 0 ? 0 : Math.sqrt(sumSq / n), p99, max };
}

async function forceAfter(
    ctx: Awaited<ReturnType<typeof createNodeGpuContext>>,
    s: ReturnType<typeof fixture>["snapshot"],
    start: Float32Array,
    tuning: GpuLayoutTuning,
    upTo: string,
): Promise<Float64Array> {
    ctx.debug.inspect = true;
    const sim = createForceAtlas2(ctx, { ...BASE, ...tuning });
    if (!(sim instanceof ForceSimulation)) {
        throw new Error("not a ForceSimulation");
    }
    const fa2 = sim as Sim;
    try {
        fa2.load(s, Float32Array.from(start));
        if (fa2.debugRunStages === undefined || fa2.inspect === undefined) {
            throw new Error("inspect not honoured");
        }
        await fa2.debugRunStages(upTo);
        const force = await fa2.inspect("force");
        return Float64Array.from(force);
    } finally {
        fa2.dispose();
    }
}

async function main(): Promise<void> {
    const ctx = await createNodeGpuContext();
    console.log(
        `adapter ${ctx.caps.vendor} / ${ctx.caps.architecture} software=${ctx.caps.software} profiler=${ctx.profiler.enabled}`,
    );
    const timed = snapshotOf(randomEdges(TIMED_NODES, 10 * TIMED_NODES, 1), { label: "knobs/random" });
    const clumpy = fixture("clumpy1000", PARITY_SCALE);
    const isolated = fixture("isolated", ISOLATED_SCALE);
    const isolatedStart = seededScenePositions(isolated.snapshot, BASE.seed ?? null, 2, 1, BASE.center ?? null);
    const clumpyStart =
        clumpy.positions ?? seededScenePositions(clumpy.snapshot, BASE.seed ?? null, 2, 1, BASE.center ?? null);
    const exactClumpy = await forceAfter(
        ctx,
        clumpy.snapshot,
        clumpyStart,
        { repulsion: "exact", compat: "paper" },
        "K3",
    );
    const exactIsolated = await forceAfter(
        ctx,
        isolated.snapshot,
        isolatedStart,
        { repulsion: "exact", compat: "paper" },
        "K3",
    );
    console.log(
        `timed graph n=${timed.nodeCount} arcs=${timed.arcCount}; clumpy1000 n=${clumpy.snapshot.nodeCount}; isolated n=${isolated.snapshot.nodeCount}`,
    );
    console.log(
        "| gridMax2D | extentFactor | 1M grid ms/iteration (mean of 20, profiler) | clumpy1000 rms / p99 | isolated rms / p99 |",
    );
    console.log("| --- | --- | --- | --- | --- |");
    for (const gridMax2D of GRID_MAX_2D) {
        for (const extentFactor of EXTENT_FACTOR) {
            const tuning: GpuLayoutTuning = {
                repulsion: "grid",
                deterministic: true,
                compat: "paper",
                gridMax2D,
                extentFactor,
            };
            ctx.debug.inspect = false;
            const sim = createForceAtlas2(ctx, {
                ...BASE,
                seed: 1,
                maxIter: 1_000_000,
                maxInFlight: 1,
                ...tuning,
            });
            let ms = Number.NaN;
            try {
                sim.load(timed, seededScenePositions(timed, 1, 2, 1, null));
                await warmClock(sim, WARM_MS);
                let sum = 0;
                let count = 0;
                for (let i = 0; i < KNOB_ITERATIONS; i++) {
                    sim.reheat();
                    await sim.step(1);
                    const v = sim.stats.msPerIteration;
                    if (v !== null) {
                        sum += v;
                        count++;
                    }
                }
                ms = count > 0 ? sum / count : Number.NaN;
            } finally {
                sim.dispose();
            }
            const c = flooredRelError(
                await forceAfter(ctx, clumpy.snapshot, clumpyStart, tuning, "G7"),
                exactClumpy,
                FLOOR_FRACTION,
            );
            const i = flooredRelError(
                await forceAfter(ctx, isolated.snapshot, isolatedStart, tuning, "G7"),
                exactIsolated,
                FLOOR_FRACTION,
            );
            console.log(
                `| ${gridMax2D} | ${extentFactor} | ${ms.toFixed(3)} | ${c.rms.toExponential(3)} / ${c.p99.toExponential(3)} | ${i.rms.toExponential(3)} / ${i.p99.toExponential(3)} |`,
            );
        }
    }
    ctx.release(timed);
    ctx.release(clumpy.snapshot);
    ctx.release(isolated.snapshot);
    ctx.dispose();
}

main().catch((error: unknown) => {
    console.error(error);
    process.exit(1);
});
